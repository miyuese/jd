```typescript
  // 2. 构建提示词
  const prompt = `你是一个面试辅导专家。请根据以下已确认的项目卡片内容，生成一段口语化的1分钟项目介绍（约150-180字）。

项目卡片信息：
- 项目背景：${card.background}
- 我的职责：${card.responsibility}
- 关键决策：${card.decisions?.join('; ')}
- 项目结果：${card.results}
- 个人成长：${card.growth}

要求：
1. 口语化表达，适合当面讲述
2. 突出与岗位匹配的核心价值
3. 结构：背景→行动→结果→价值
4. 字数控制在150-180字
5. 不要使用书面化的"负责"、"参与"等词汇，改用口语表达
6. 只输出讲稿内容，不要加任何解释`

  // 3. 调用AI生成
  const { text } = await generateText({
    model,
    prompt
  })

  // 4. 保存版本（如需要）
  // await prisma.interviewVersion.create({...})

  return { content: text }
}
```

```typescript
import { auth } from '@clerk/nextjs/server'
import { prisma } from '@/lib/prisma'

export async function saveOutputVersion(
  projectId: string,
  type: string,
  content: string,
  sourceCardVersionId?: string,
  sourceMatchVersionId?: string
) {
  const { userId } = await auth()
  if (!userId) throw new Error('未登录')

  // 获取当前最大版本号
  const maxVersion = await prisma.outputVersion.findFirst({
    where: { projectId, userId, type },
    orderBy: { version: 'desc' }
  })

  const version = (maxVersion?.version || 0) + 1

  const outputVersion = await prisma.outputVersion.create({
    data: {
      projectId,
      userId,
      type,
      content,
      sourceCardVersionId,
      sourceMatchVersionId,
      version
    }
  })

  return { success: true, version: outputVersion }
}
```

```typescript
    }),
    prisma.outputVersion.findMany({

      orderBy: { createdAt: 'desc' }
    })
  ])

  // 合并并排序
  const allVersions = [
    ...cardVersions.map(v => ({ ...v, versionType: 'card' as const })),
    ...matchVersions.map(v => ({ ...v, versionType: 'match' as const })),
    ...outputVersions.map(v => ({ ...v, versionType: 'output' as const }))

  return allVersions
}
```

```typescript
// app/api/upload/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { auth } from '@clerk/nextjs/server'
import { put } from '@vercel/blob'

export async function POST(request: NextRequest) {
  const { userId } = await auth()
  if (!userId) {
    return NextResponse.json({ error: '未登录' }, { status: 401 })
  }

  const formData = await request.formData()
  const file = formData.get('file') as File

  if (!file) {
    return NextResponse.json({ error: '未找到文件' }, { status: 400 })
  }

  // 文件类型校验
  const allowedTypes = [
    'application/vnd.openxmlformats-officedocument.wordprocessingml.document', // docx
    'application/pdf'
  ]
  
  if (!allowedTypes.includes(file.type)) {
    return NextResponse.json({ 
      error: '只支持 .docx 和 .pdf 格式文件' 
    }, { status: 400 })
  }

  // 文件大小限制（5MB）
  if (file.size > 5 * 1024 * 1024) {
    return NextResponse.json({ 
      error: '文件大小不能超过5MB' 
    }, { status: 400 })
  }

  try {
    // 上传到 Vercel Blob
    const blob = await put(`materials/${userId}/${Date.now()}-${file.name}`, file, {
      access: 'public',
    })

    return NextResponse.json({
      success: true,
      url: blob.url,
      fileName: file.name,
      fileType: file.type,
      size: file.size
    })
  } catch (error) {
    return NextResponse.json({ 
      error: '上传失败，请重试' 
    }, { status: 500 })
  }
}
```

```typescript
// app/api/parse/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { auth } from '@clerk/nextjs/server'
import mammoth from 'mammoth'
import pdf from 'pdf-parse'

export async function POST(request: NextRequest) {
  const { userId } = await auth()
  if (!userId) {
    return NextResponse.json({ error: '未登录' }, { status: 401 })
  }

  const { fileUrl, fileType } = await request.json()

  if (!fileUrl || !fileType) {
    return NextResponse.json({ error: '缺少参数' }, { status: 400 })
  }

  try {
    // 下载文件
    const fileResponse = await fetch(fileUrl)
    const arrayBuffer = await fileResponse.arrayBuffer()
    const buffer = Buffer.from(arrayBuffer)

    let text = ''

    if (fileType.includes('wordprocessingml')) {
      // 解析 docx
      const result = await mammoth.extractRawText({ buffer })
      text = result.value
    } else if (fileType === 'application/pdf') {
      // 解析 pdf
      const data = await pdf(buffer)
      text = data.text
      
      // 检查是否为扫描版（文本过少）
      if (text.trim().length < 50) {
        return NextResponse.json({ 
          error: '该PDF为扫描版或图片型PDF，无法提取文字内容，请手动粘贴文本' 
        }, { status: 400 })
      }
    }

    if (!text.trim()) {
      return NextResponse.json({ 
        error: '无法提取文件内容，请手动粘贴文本' 
      }, { status: 400 })
    }

    return NextResponse.json({ success: true, text })
  } catch (error) {
    return NextResponse.json({ 
      error: '解析失败，请手动粘贴文本内容' 
    }, { status: 500 })
  }
}
```

```typescript
}: { 
  value: string, 
  onChange: (text: string) => void 
}) {
  const [parsing, setParsing] = useState(false)

  async function handleUploadSuccess(fileInfo: any) {
    setParsing(true)
    try {
      const response = await fetch('/api/parse', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          fileUrl: fileInfo.url,
          fileType: fileInfo.fileType
        })
      })

      const result = await response.json()

      if (!response.ok) {
        throw new Error(result.error)
      }

      // 回填到材料区
      onChange(result.text)
    } catch (error: any) {
      alert(error.message || '解析失败，请手动粘贴')
    } finally {
      setParsing(false)
    }
  }

  return (
    <div className="space-y-4">
      <div className="flex items-center justify-between">
        <h3 className="text-lg font-semibold">上传文件</h3>
        <FileUpload onUploadSuccess={handleUploadSuccess} />
      </div>
      {parsing && <p className="text-sm text-muted-foreground">解析中...</p>}
      <textarea
        className="w-full min-h-[200px] p-4 border rounded-md"
        value={value}
        onChange={(e) => onChange(e.target.value)}
        placeholder="粘贴项目材料内容，或上传 .docx/.pdf 文件自动提取..."
      />
    </div>
  )
}
```

```typescript
// lib/error-handler.ts
export function getErrorMessage(error: any): string {
  if (typeof error === 'string') return error
  
  // AI接口错误
  if (error?.message?.includes('API') || error?.status === 429) {
    return 'AI服务暂时不可用，请稍后重试'
  }
  
  // 数据库错误
  if (error?.code?.startsWith('P')) {
    return '数据操作失败，请刷新后重试'
  }
  
  // 文件上传错误
  if (error?.message?.includes('上传')) {
    return error.message
  }
  
  return error?.message || '操作失败，请重试'
}
```

```javascript
// 简单的线上验证脚本
const https = require('https')

const CHECKS = [
  { name: '首页加载', url: '/' },
  { name: '登录页', url: '/sign-in' },
  { name: 'API健康检查', url: '/api/health' } // 需要自己实现
]

async function checkPage(baseUrl, path) {
  return new Promise((resolve) => {
    https.get(`${baseUrl}${path}`, (res) => {
      resolve({
        status: res.statusCode,
        ok: res.statusCode >= 200 && res.statusCode < 400
      })
    }).on('error', () => resolve({ status: 0, ok: false }))
  })
}

async function main() {
  const baseUrl = process.env.VERCEL_URL || 'https://your-app.vercel.app'
  
  console.log(`验证线上环境: ${baseUrl}\n`)
  
  for (const check of CHECKS) {
    const result = await checkPage(baseUrl, check.url)
    console.log(`${result.ok ? '✓' : '✗'} ${check.name}: ${result.status}`)
  }
}

main()
```
